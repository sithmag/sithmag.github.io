# GUI
"""
Calls a series of pySimpleGUI demonstration functions and collects
output for use in GUI lab.
Author: Sithma
Date: 6/9/23
"""
import PySimpleGUI as sg
import numpy as np
from tkinter import *
from matplotlib.backends.backend_tkagg import FigureCanvasTkAgg
from matplotlib.figure import Figure
import datetime
import serial
import time
import sys
import csv
import re

portName = "/dev/tty.ESP32sithma-ESP32SPP"

serialPort = serial.Serial(
    port=portName,
    baudrate=115200,
    bytesize=8,
    timeout=2,
    stopbits=serial.STOPBITS_ONE
)

serialPort.close()

try:
    serialPort.open()
except:
    print("Port open failed: " + portName)
    for e in sys.exc_info():
        print("  ", e)

HeartRate = 0

def interpretData():
    """
    Reads and interprets pulse data from the microcontroller.

    Outputs:
    PulseData: list of pulse readings
    HeartRate: averaged heart rate value
    no_data_check: whether data is being received
    sequence: packet number from ESP
    button: board button status
    """
    HeartRate = 0
    PulseData = []
    sequence = 0
    button = 0

    if serialPort.isOpen():
        if serialPort.in_waiting > 0:
            serial_string = serialPort.readline().decode("ASCII").strip()
            unpacked_data = serial_string.split(",")
            data_points = []

            if unpacked_data[0] == "No data":
                PulseData = [0] * 50
                HeartRate = 0
                no_data_check = False
                sequence = 0
            else:
                no_data_check = True

                for i in range(0, 53):
                    input_string = unpacked_data[i]
                    numeric_string = re.sub(r'[^0-9.]', '', input_string)

                    if 0 <= i < 50:
                        try:
                            value = int(numeric_string)
                        except ValueError:
                            value = 0
                        data_points.append(value)

                    if i == 50:
                        HeartRate = float(numeric_string)

                    if i == 51:
                        sequence = int(numeric_string)

                    if i == 52:
                        button = int(numeric_string)

                PulseData = data_points
        else:
            PulseData = False
            no_data_check = False
    else:
        PulseData = False
        no_data_check = False

    return PulseData, HeartRate, no_data_check, sequence, button
